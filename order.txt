// Using node-cron
import cron from 'node-cron';

cron.schedule('0 2 * * *', async () => {
  try {
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - 30);
    
    const result = await prisma.$transaction(async (tx) => {
      const oldCarts = await tx.cart.findMany({
        where: { updatedAt: { lt: cutoffDate } },
        select: { id: true }
      });
      
      await tx.cartItem.deleteMany({
        where: { cartId: { in: oldCarts.map(c => c.id) } }
      });
      
      return oldCarts.length;
    });
    
    console.log(`Cleaned up ${result} abandoned carts`);
  } catch (error) {
    console.error('Cart cleanup failed:', error);
  }
});
